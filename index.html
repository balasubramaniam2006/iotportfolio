Sensor Interfacing & Environmental Monitoring

Multi-bus acquisition using DHT11/22 (Single-Wire) and SSD1306 (I2C 0x3C)

Architecture & Wiring

Acquires calibrated ambient temperature, relative humidity, and light intensity (Lux) via voltage divider ADC. Data is serialized for local I2C display and buffered for cloud dispatch.

Task1_DHT_OLED_Interfacing.ino 
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <DHT.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

#define DHTPIN 4
#define DHTTYPE DHT22
#define LDRPIN 34

DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(115200);
  dht.begin();
  
  if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println(F("SSD1306 allocation failed"));
    for(;;);
  }
  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  display.setTextSize(1);
}

void loop() {
  float h = dht.readHumidity();
  float t = dht.readTemperature();
  int rawLDR = analogRead(LDRPIN);
  float lux = map(rawLDR, 0, 4095, 0, 1000);

  if (isnan(h) || isnan(t)) {
    Serial.println("Failed reading sensor!");
    return;
  }

  display.clearDisplay();
  display.setCursor(0, 0);
  display.println("SYS STATUS: OK");
  display.printf("Temp: %.1f C\n", t);
  display.printf("Humidity: %.1f %%\n", h);
  display.printf("Lux: %.0f lx\n", lux);
  display.display();

  delay(2000); // 2-second telemetry cycle
}

Actuator Control & High-Load Relays

Galvanic isolation, flyback diode protection, active-low relay coil triggering

Virtual Relay Interface

Test optocoupler activation states. High-power AC mains loads (220V/10A) and DC fans are routed via isolated NO/NC contacts.

Load Channel 1 (Bulb)

GPIO 26 · Opto-Isolated

Load Channel 2 (Exhaust Fan)

Safety Protocol: Snubber circuits (RC 100Ω + 0.1µF) fitted parallel to inductive relay contacts to suppress back-EMF arcing.
Task2_Relay_Actuator_Switch.ino 
const int RELAY_BULB_PIN = 26; // Channel 1 (Active LOW)
const int RELAY_FAN_PIN  = 27; // Channel 2 (Active LOW)

void setLoadState(int pin, bool state) {
  // Most 5V/3.3V optocoupler modules activate on LOW level
  digitalWrite(pin, state ? LOW : HIGH);
}

void setup() {
  Serial.begin(115200);
  
  pinMode(RELAY_BULB_PIN, OUTPUT);
  pinMode(RELAY_FAN_PIN, OUTPUT);

  // Initialize relays in OFF position safely
  digitalWrite(RELAY_BULB_PIN, HIGH);
  digitalWrite(RELAY_FAN_PIN, HIGH);
  
  Serial.println("Actuator Controller Online. Awaiting command pulses...");
}

void loop() {
  if (Serial.available() > 0) {
    char cmd = Serial.read();
    if (cmd == '1') {
      setLoadState(RELAY_BULB_PIN, true);
      Serial.println("CH1 BULB: ENGAGED");
    } else if (cmd == '0') {
      setLoadState(RELAY_BULB_PIN, false);
      Serial.println("CH1 BULB: DISENGAGED");
    } else if (cmd == '2') {
      setLoadState(RELAY_FAN_PIN, true);
      Serial.println("CH2 FAN: ENGAGED");
    } else if (cmd == '3') {
      setLoadState(RELAY_FAN_PIN, false);
      Serial.println("CH2 FAN: DISENGAGED");
    }
  }
}

#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>

const char* ssid = "WIFI_SSID";
const char* password = "WIFI_PASSWORD";
const char* mqtt_server = "broker.emqx.io";

WiFiClient espClient;
PubSubClient client(espClient);

void setup_wifi() {
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi connected, IP: " + WiFi.localIP().toString());
}

void reconnect() {
  while (!client.connected()) {
    String clientId = "ESP32Client-" + String(random(0xffff), HEX);
    if (client.connect(clientId.c_str())) {
      client.subscribe("smartyy/devices/control");
    } else {
      delay(3000);
    }
  }
}

void publishTelemetry(float temp, float hum, int light) {
  StaticJsonDocument<200> doc;
  doc["temperature"] = temp;
  doc["humidity"] = hum;
  doc["light_lux"] = light;
  doc["rssi"] = WiFi.RSSI();

  char buffer[256];
  serializeJson(doc, buffer);
  client.publish("smartyy/telemetry/nodes", buffer);
}

void setup() {
  Serial.begin(115200);
  setup_wifi();
  client.setServer(mqtt_server, 1883);
}

void loop() {
  if (!client.connected()) reconnect();
  client.loop();
  // Call publishTelemetry periodic loop via non-blocking millis()
}

Task4_Gas_Safety_Cutoff.ino 
const int MQ2_ANALOG_PIN  = 35; // ADC Pin for Smoke / LPG
const int BUZZER_PWM_PIN  = 18; // Piezo PWM channel
const int SAFETY_RELAY_PIN = 19; // Main contactor cut-off
const int THRESHOLD_PPM   = 350;

void setup() {
  Serial.begin(115200);
  pinMode(SAFETY_RELAY_PIN, OUTPUT);
  digitalWrite(SAFETY_RELAY_PIN, LOW); // Active State

  ledcSetup(0, 2400, 8); // Channel 0, 2.4kHz alarm freq
  ledcAttachPin(BUZZER_PWM_PIN, 0);
  ledcWrite(0, 0);       // Buzzer Silent
}

void loop() {
  int rawGas = analogRead(MQ2_ANALOG_PIN);
  float ppm = map(rawGas, 0, 4095, 50, 1000);

  if (ppm >= THRESHOLD_PPM) {
    // 1. Trigger safety interlock (Trip Relay immediately)
    digitalWrite(SAFETY_RELAY_PIN, HIGH);
    
    // 2. Modulate audio-visual alarm
    ledcWrite(0, 180);
    
    // 3. Dispatch critical alert to cloud API endpoint
    Serial.printf("[ALERT CRITICAL] Gas Leak: %.1f PPM. Tripping Relays!\n", ppm);
  } else {
    digitalWrite(SAFETY_RELAY_PIN, LOW);
    ledcWrite(0, 0);
  }
  delay(500);
}

Edge Computing & Hardware State Engine

Deterministic Finite State Machine (FSM) with deadband hysteresis & offline continuity

FSM Execution Matrix

Prevents relay chattering through a 2.5°C hysteresis band and guarantees local autonomy during Internet disconnections.

STATE: MONITOR_IDLE T < 28°C · Relays Rest State · Sleep Mode
STATE: ACTIVE_COOLING T ≥ 30.5°C · Exhaust Fan ENGAGED
STATE: OFFLINE_FALLBACK WiFi Loss > 15s · Local Flash Rules Active
Task5_Edge_Finite_State_Machine.ino 
enum SystemState { STATE_IDLE, STATE_COOLING, STATE_EMERGENCY, STATE_OFFLINE };
SystemState currentState = STATE_IDLE;

const float TEMP_UPPER_LIMIT = 30.5; // Fan Engage
const float TEMP_LOWER_LIMIT = 28.0; // Fan Disengage (Hysteresis 2.5C)

void processStateMachine(float currentTemp, bool wifiConnected) {
  switch (currentState) {
    case STATE_IDLE:
      if (currentTemp >= TEMP_UPPER_LIMIT) {
        digitalWrite(27, LOW); // Relay Fan ON
        currentState = STATE_COOLING;
        Serial.println("FSM -> STATE_COOLING");
      }
      break;

    case STATE_COOLING:
      if (currentTemp <= TEMP_LOWER_LIMIT) {
        digitalWrite(27, HIGH); // Relay Fan OFF
        currentState = STATE_IDLE;
        Serial.println("FSM -> STATE_IDLE");
      }
      break;

    case STATE_OFFLINE:
      // Local fallback algorithm operates without waiting for cloud commands
      if (currentTemp >= TEMP_UPPER_LIMIT) digitalWrite(27, LOW);
      if (wifiConnected) currentState = STATE_IDLE;
      break;
  }
}
